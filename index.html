<script>

/* =======================

   STONE RENDERER — JS

   FSM 0–7, deterministic

   Oracle call after HALT

   ======================= */


const STATE={DORMANT:0,WARNING:1,PRIME:2,COMMIT:3,INPUT:4,HALT:5,RESPONSE:6,TERMINAL:7};

let state=STATE.DORMANT;

let question="", answer="";


const CONTENT={

  s1:{t:"Notice",b:"This is a threshold.\n\nYou may leave now with no loss.\n\nIf you continue:\n- reflection, not comfort\n- no optimisation for relief",x:"Exit",c:"Continue"},

  s2:{t:"Prime",b:"This does not answer you.\nIt shows you.\n\nSpeed will not help you here.",x:"Exit",c:"Enter"},

  s3:{t:"Threshold",b:"Proceed knowing you cannot control what reflects.",x:"Exit",c:"Proceed"},

  s5:{t:"Last brake",b:"Final chance to refuse.\n\nContinue = loss of narrative control.",x:"Exit",c:"Continue"}

};


const ORACLE = {

  mode: "remote",          // "remote" | "local"

  endpoint: "/api/oracle", // ✅ CHANGE: GitHub Pages / same-origin default (use your Worker URL if different domain)

  timeoutMs: 15000,

  version: "oracle-i/1.0"

};


const app=document.getElementById("app"),

ov=document.getElementById("overlay"),

ovT=document.getElementById("ovTitle"),

ovB=document.getElementById("ovBody"),

ovC=document.getElementById("ovControls"),

wake=document.getElementById("wake"),

inp=document.getElementById("inputPane"),

qEl=document.getElementById("question"),

respP=document.getElementById("responsePane"),

resp=document.getElementById("response");


const submitBtn=document.getElementById("submitQ");

const exitInputBtn=document.getElementById("exitInput");

const endBtn=document.getElementById("end");


function setState(s){

  state=s; app.className="state-"+s;

  ov.classList.add("hidden"); inp.classList.add("hidden"); respP.classList.add("hidden");

  wake.classList.toggle("hidden", s!==STATE.DORMANT);


  if(s===STATE.WARNING) return render(CONTENT.s1,()=>setState(STATE.PRIME));

  if(s===STATE.PRIME)   return render(CONTENT.s2,()=>setState(STATE.COMMIT));

  if(s===STATE.COMMIT)  return render(CONTENT.s3,()=>setState(STATE.INPUT));


  if(s===STATE.INPUT){

    inp.classList.remove("hidden");

    qEl.focus();

    return;

  }


  if(s===STATE.HALT)    return render(CONTENT.s5,reflect);


  if(s===STATE.RESPONSE){

    resp.textContent=answer;

    return respP.classList.remove("hidden");

  }


  if(s===STATE.TERMINAL){

    submitBtn.disabled = true;

    exitInputBtn.disabled = true;

    endBtn.disabled = true;

    return render({t:"Ended",b:"Session closed.",x:null,c:"Restart"},reset);

  }

}


function render(c,onContinue){

  ovT.textContent=c.t; ovB.textContent=c.b; ovC.innerHTML="";

  if(c.x) ovC.appendChild(btn(c.x,()=>setState(STATE.TERMINAL),true));

  ovC.appendChild(btn(c.c,onContinue,false));

  ov.classList.remove("hidden");

}


function btn(label,fn,danger){

  const b=document.createElement("button");

  b.textContent=label;

  b.className="btn"+(danger?" danger":" primary");

  b.onclick=fn;

  return b;

}


async function callOracle(q){

  if(ORACLE.mode === "local"){

    return [

      "DIAGNOSTIC",

      "",

      "You asked:",

      q,

      "",

      "Reflection:",

      "- What do you assume must be controllable?",

      "- Where is judgement outsourced to speed?",

      "",

      "Room-Freeze:",

      "Can any operator halt this at the same speed/friction it executes?"

    ].join("\n");

  }


  if(!ORACLE.endpoint){

    throw new Error("Oracle endpoint not set.");

  }


  const controller = new AbortController();

  const timer = setTimeout(()=>controller.abort(), ORACLE.timeoutMs);


  try{

    const r = await fetch(ORACLE.endpoint, {

      method:"POST",

      headers:{

        "Content-Type":"application/json",

        "X-Oracle-Version": ORACLE.version

      },

      body: JSON.stringify({ question: q, client: { renderer:"looking-glass", fsm:"0-7" } }),

      signal: controller.signal

    });


    if(!r.ok){

      const t = await r.text().catch(()=> "");

      throw new Error(`Oracle error (${r.status}): ${t || "no body"}`);

    }


    const data = await r.json();

    if(!data || typeof data.answer !== "string"){

      throw new Error("Malformed oracle response.");

    }


    return data.answer;

  } finally {

    clearTimeout(timer);

  }

}


async function reflect(){

  // close race window while in-flight

  submitBtn.disabled = true;

  exitInputBtn.disabled = true;

  endBtn.disabled = true;


  ovT.textContent = "Calling Oracle";

  ovB.textContent = "Hold.\n\nNo retries.\nNo silent fallback.";

  ovC.innerHTML = "";

  ovC.appendChild(btn("Abort", ()=>setState(STATE.TERMINAL), true));

  ov.classList.remove("hidden");


  try{

    answer = await callOracle(question);

  } catch(e){

    answer = [

      "ORACLE FAILURE",

      "",

      String(e && e.message ? e.message : e),

      "",

      "Room-Freeze:",

      "Can any operator halt this at the same speed/friction it executes?"

    ].join("\n");

  }


  setState(STATE.RESPONSE);

}


function reset(){

  question=""; answer="";

  qEl.value="";

  submitBtn.disabled = false;

  exitInputBtn.disabled = false;

  endBtn.disabled = false;

  setState(STATE.DORMANT);

}


wake.onclick=()=> state===STATE.DORMANT && setState(STATE.WARNING);


submitBtn.onclick=()=>{

  const v=qEl.value.trim();

  if(!v) return;

  question=v;

  setState(STATE.HALT);

};


exitInputBtn.onclick=()=>setState(STATE.TERMINAL);

endBtn.onclick=()=>setState(STATE.TERMINAL);

</script>
